% Shared by every driver. \ThesisCoverTypeLabel (config/thesis-language.tex)
% prints PRA-SKRIPSI, SKRIPSI (DRAF BAB I--IV), or SKRIPSI depending on
% \ThesisTarget. Every other line comes from config/thesis-config.tex.
%
% The two \hspace* values pull the logo and the yellow rule out of the text
% block: the rule has to bleed to both paper edges, which the geometry margins
% would otherwise prevent.
%
% The outer (orange) cover and the inner (white) cover share one layout, so the
% layout is written once as \ThesisCoverBody.
\newcommand{\ThesisCoverBody}{%
  \begin{flushleft}
    {\large\bfseries \MakeUppercase{\ThesisCoverTypeLabel}\par}
  \end{flushleft}
  \vspace{0.2cm}
  \begin{flushleft}{\Large\bfseries \ThesisTitleUpper\par}\end{flushleft}
  \vspace{1.2cm}
  \begin{flushleft}
    {\large\bfseries \StudentNameUpper\par}
    {\large \NpmLabel\ \StudentID\par}
  \end{flushleft}
  \vspace{1.2cm}
  \begin{flushleft}
    {\large\bfseries \AdvisorBlockLabel\par}
    {\large \AdvisorOne\par}
    {\large \AdvisorTwo\par}
  \end{flushleft}
  \vspace{1.2cm}
  \begin{flushleft}
    {\normalsize\bfseries \MinistryLabel\par}
    {\normalsize\bfseries \UniversityLabel\par}
    {\normalsize\bfseries \MakeUppercase{\FacultyLabel}\par}
    {\normalsize\bfseries \ProgramLabel\par}
    {\normalsize\bfseries SURABAYA\par}
    {\normalsize\bfseries \SubmissionYear\par}
  \end{flushleft}%
}

% ── OUTER COVER (orange), final manuscript only by default
% The background is drawn as a TikZ overlay before the logo: full orange page,
% a white band from the top edge to 1 cm, an orange band from 1 to 2 cm, and a
% white area behind the logo that ends inside the yellow rule, so the rule
% printed afterwards hides the boundary. Text on this cover is white.
% hyperref page anchors are switched off so this unnumbered page does not clash
% with the inner cover's page number.
\ifThesisColorCover
\hypersetup{pageanchor=false}
\begin{titlepage}
  \renewcommand{\familydefault}{\sfdefault}\sffamily
  \thispagestyle{empty}
  \begin{flushleft}
    \noindent\begin{tikzpicture}[remember picture, overlay]
      \fill[coverOrange] (current page.south west) rectangle (current page.north east);
      \fill[white] (current page.north west) rectangle ([yshift=-1cm]current page.north east);
      \fill[white] ([yshift=-2cm]current page.north west) rectangle ([yshift=-7.85cm]current page.north east);
    \end{tikzpicture}%
    \hspace*{-\dimexpr\oddsidemargin+0.2cm\relax}\includegraphics[width=4cm]{logo_upn.png}
  \end{flushleft}
  \vspace{0.35cm}
  \noindent\hspace*{-\dimexpr\oddsidemargin+1in\relax}{\color{coverYellow}\rule{\paperwidth}{18pt}}
  \vspace{0.45cm}
  \color{white}
  \ThesisCoverBody
\end{titlepage}
\hypersetup{pageanchor=true}

% blank page behind it: the book block starts at the inner cover.

\fi
